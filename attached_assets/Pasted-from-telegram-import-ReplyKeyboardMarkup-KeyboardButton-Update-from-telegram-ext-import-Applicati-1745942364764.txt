# RSI Göstergesi Al ve Sinyal Gönder
async def rsi_signal(update: Update, context: ContextTypes.DEFAULT_TYPE):
    coin = context.user_data.get('coin', 'BTCUSDT')
    period = context.user_data.get('rsi_period', 14)  # RSI dönemi 14 varsayılan

    rsi = await get_rsi(coin, period)
    if rsi > 70:
        signal = "📉 Overbought! Satış yapmayı düşünebilirsiniz."
    elif rsi < 30:
        signal = "📈 Oversold! Alış yapmayı düşünebilirsiniz."
    else:
        signal = "📊 RSI normal seviyelerde."

    await update.message.reply_text(f"RSI değeri: {rsi}\n{signal}")
    return ConversationHandler.END


# RSI Hesaplama (Binance API'sinden veri çekme)
async def get_rsi(symbol: str, period: int):
    url = f'https://api.binance.com/api/v3/klines'
    params = {
        'symbol': symbol,
        'interval': '1h',  # 1 saatlik veriler
        'limit': period + 1  # RSI için bir fazla veri lazım
    }
    response = requests.get(url, params=params)
    data = response.json()
    closes = [float(kline[4]) for kline in data]  # Kapanış fiyatları

    gains = []
    losses = []
    for i in range(1, len(closes)):
        change = closes[i] - closes[i - 1]
        if change > 0:
            gains.append(change)
            losses.append(0)
        else:
            losses.append(abs(change))
            gains.append(0)

    avg_gain = sum(gains) / period
    avg_loss = sum(losses) / period

    if avg_loss == 0:
        rsi = 100
    else:
        rs = avg_gain / avg_loss
        rsi = 100 - (100 / (1 + rs))

    return rsi
